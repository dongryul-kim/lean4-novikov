\section{Novikov descent}

We now define the notion of a Novikov descent datum and prove that the category
of Novikov descent data is equivalent to the category of finite projective
modules.

\subsection{Abstract descent data}

\begin{definition}
  \label{def:cosimplicial_ring}
  \lean{Novikov.Descent.Abstract.CosimplicialRing}
  \leanok
  A \emph{cosimplicial ring} (truncated to levels $1$--$3$) consists of
  three commutative rings $R_1, R_2, R_3$ with face maps
  $\pi_1^\ast, \pi_2^\ast \colon R_1 \to R_2$ and
  $\pi_{12}^\ast, \pi_{13}^\ast, \pi_{23}^\ast \colon R_2 \to R_3$,
  defining composites $\rho_i^\ast = \pi_{\cdot}^\ast\pi_{\cdot}^\ast \colon R_1 \to R_3$
  satisfying the six cosimplicial identities.
  For a ring homomorphism $\pi^\ast \colon R \to S$, we write
  $\pi^\ast M = S \otimes_{R, \pi^\ast} M$ for the base change of an $R$-module $M$.
\end{definition}

\begin{definition}
  \label{def:descent_datum}
  \lean{Novikov.Descent.Abstract.DescentDatum}
  \leanok
  \uses{def:cosimplicial_ring}
  A \emph{descent datum} over a cosimplicial ring $C$ is a finite projective
  $R_1$-module $M$ with an $R_2$-linear isomorphism
  $\varphi \colon \pi_1^\ast M \xrightarrow{\sim} \pi_2^\ast M$ satisfying the
  cocycle condition
  $\pi_{23}^\ast\varphi \circ \pi_{12}^\ast\varphi = \pi_{13}^\ast\varphi$.
  Morphisms are $R_1$-linear maps commuting with $\varphi$ after base change;
  they form a category with an internal Hom.
\end{definition}

\begin{definition}
  \label{def:extended_cosimplicial_ring}
  \lean{Novikov.Descent.Abstract.ExtendedCosimplicialRing}
  \leanok
  \uses{def:cosimplicial_ring}
  An \emph{extended cosimplicial ring} adds a ring $R_0$ and
  $\pi_0^\ast \colon R_0 \to R_1$ with
  $\pi_1^\ast\pi_0^\ast = \pi_2^\ast\pi_0^\ast$.
\end{definition}

\begin{definition}
  \label{def:constant_descent_datum}
  \lean{Novikov.Descent.Abstract.constantDescentDatum}
  \leanok
  \uses{def:extended_cosimplicial_ring, def:descent_datum}
  For a finite projective $R_0$-module $M$, the \emph{constant descent datum}
  $\operatorname{Const}(M)$ has underlying $R_1$-module $R_1 \otimes_{R_0} M$,
  with $\varphi$ the composition of associativity isomorphisms
  $\pi_1^\ast(R_1 \otimes_{R_0} M) \cong R_2 \otimes_{R_0} M
   \cong \pi_2^\ast(R_1 \otimes_{R_0} M)$
  via $\pi_1^\ast\pi_0^\ast = \pi_2^\ast\pi_0^\ast$.
\end{definition}

\begin{definition}
  \label{def:constant_descent_functor}
  \lean{Novikov.Descent.Abstract.constantDescentDatumFunctor}
  \leanok
  \uses{def:constant_descent_datum}
  This defines a functor
  $\operatorname{Const} \colon \mathsf{Vect}(R_0) \to \mathsf{Desc}(E)$.
  It preserves internal Hom:
  $\underline{\operatorname{Hom}}(\operatorname{Const}(M), \operatorname{Const}(N))
   \cong \operatorname{Const}(\operatorname{Hom}_{R_0}(M, N))$.
\end{definition}

\begin{theorem}[Abstract full faithfulness of constant descent]
  \label{thm:constant_descent_fully_faithful}
  \lean{Novikov.Descent.Abstract.constantDescentDatumFunctor_fullyFaithful}
  \leanok
  \uses{def:constant_descent_functor}
  If the natural map $R_0 \to \operatorname{eq}(\pi_1^\ast, \pi_2^\ast)$ is bijective,
  then $\operatorname{Const}$ is fully faithful.
\end{theorem}

\begin{proof}
  \leanok
  \textbf{Faithfulness.}  Bijectivity of $R_0 \to \operatorname{eq}(\pi_1^\ast,\pi_2^\ast)$
  implies $\pi_0^\ast \colon R_0 \to R_1$ is injective, hence
  $p \mapsto 1 \otimes p$ is injective for any finite projective $P$.
  Taking $P = \operatorname{Hom}(M, N)$, two maps $f,g$ equal after
  base change to $R_1$ satisfy $1 \otimes (f-g) = 0$, forcing $f=g$.

  \textbf{Fullness.}  A morphism $F \colon \operatorname{Const}(M) \to
  \operatorname{Const}(N)$ yields via the internal Hom isomorphism an
  element $f \in R_1 \otimes_{R_0} \operatorname{Hom}(M,N)$ fixed by
  $\varphi$.  The equalizer hypothesis plus finite projectivity show
  $f = 1 \otimes f_0$ for some $R_0$-linear map $f_0$, which recovers $F$.
\end{proof}

\subsection{Novikov descent data}

\begin{definition}[Definition~2.9]
  \label{def:novikov_descent_datum}
  \lean{Novikov.Descent.novikovCosimplicialRing}
  \leanok
  \uses{def:cosimplicial_ring, def:novikov_series, def:novikov_substitute,
    thm:novikov_ring, thm:novikov_substitute_functorial}
  Let $A$ be a commutative ring and $\Gamma \subseteq \mathbb{R}$ an additive
  submonoid.  The \emph{Novikov cosimplicial ring} has
  $R_1 = A((t^\Gamma))$,
  $R_2 = A((t^\Gamma, u^\Gamma))$,
  $R_3 = A((t^\Gamma, u^\Gamma, v^\Gamma))$,
  with face maps given by substituting the variables:
  $\pi_1^\ast, \pi_2^\ast \colon R_1 \to R_2$ send $t$ to $t$ or $u$ respectively;
  $\pi_{12}^\ast, \pi_{13}^\ast, \pi_{23}^\ast \colon R_2 \to R_3$ are the three coface
  maps between two and three variables.
  The extended version sets $R_0 = A$ with $\pi_0^\ast \colon A \to A((t^\Gamma))$
  the inclusion as constant series.
  A \emph{Novikov descent datum} is a descent datum over this cosimplicial ring.
\end{definition}

\begin{lemma}[The Novikov equalizer]
  \label{lem:novikov_equalizer}
  \lean{Novikov.Descent.novikovCosimplicialRing_equalizer,
    Novikov.Descent.novikovExtendedCosimplicialRing_equalizer_bijective}
  \leanok
  \uses{def:novikov_descent_datum}
  The equalizer of
  $\pi_1^\ast, \pi_2^\ast \colon A((t^\Gamma)) \to A((t^\Gamma, u^\Gamma))$
  is exactly the constant series $A$.
  Hence $A \to \operatorname{eq}(\pi_1^\ast, \pi_2^\ast)$ is bijective.
\end{lemma}

\begin{proof}
  \leanok
  If $f(t) \in A((t^\Gamma))$ satisfies $f(t) = f(u)$ in $A((t^\Gamma, u^\Gamma))$,
  evaluate at the exponent $(d, 0)$: the left side gives the
  $d$-coefficient of $f$, while the right side gives the
  $d$-coefficient of the series whose only variable is $u$, hence $0$ for $d \neq 0$.
  Thus all coefficients of $f$ vanish except possibly the constant term, so
  $f \in A$.
\end{proof}

\begin{proposition}[Proposition~2.12]
  \label{prop:vect_to_novikov_descent_fully_faithful}
  \lean{Novikov.Descent.vectToNovikovDescent_fullyFaithful}
  \leanok
  \uses{def:novikov_descent_datum, thm:constant_descent_fully_faithful,
    lem:novikov_equalizer}
  The constant-descent functor
  $\mathsf{Vect}(A) \to \mathsf{NovDD}(A, \Gamma)$ is fully faithful.
\end{proposition}

\begin{proof}
  \leanok
  The equalizer hypothesis of
  Theorem~\ref{thm:constant_descent_fully_faithful}
  is satisfied by Lemma~\ref{lem:novikov_equalizer}.
\end{proof}

\subsection{Isocrystal from a descent datum}

Throughout, $\Gamma = \mathbb{R}$, $A$ is a commutative ring, and $\lambda > 1$.
We construct a functor $\mathsf{NovDD}(A) \to \mathsf{NovIsoc}(A)$.

\begin{lemma}[The second-face module equalizer]
  \label{lem:pi2_module_equalizer}
  \lean{Novikov.Descent.pi2_module_equalizer_oneTmul}
  \leanok
  \uses{def:novikov_descent_datum, lem:novikov_equalizer}
  For a finite projective (hence flat) $A((t^{\mathbb{R}}))$-module $M$, the diagram
  \[
    M \xrightarrow{\pi_2^\ast} \pi_2^\ast M
      \overset{\pi_{13}^\ast}{\underset{\pi_{23}^\ast}{\rightrightarrows}}
      \rho_3^\ast M
  \]
  is an equalizer: every $x \in \pi_2^\ast M$ with $\pi_{13}^\ast x = \pi_{23}^\ast x$
  equals $1 \otimes m$ for a unique $m \in M$.
\end{lemma}

\begin{proof}
  \leanok
  Tensoring with the flat $M$ preserves exactness, so it suffices to treat
  $M = A((t^{\mathbb{R}}))$, where the claim is the ring-level equalizer of
  $\pi_2^\ast$ against $\pi_{13}^\ast - \pi_{23}^\ast$ (a variant of
  Lemma~\ref{lem:novikov_equalizer}).
\end{proof}

\begin{definition}
  \label{def:face_frobenius}
  \lean{Novikov.Descent.F2, Novikov.Descent.F3}
  \leanok
  \uses{def:novikov_descent_datum, def:coordinate_frobenius}
  The $A$-algebra automorphisms scaling one variable by $\lambda$, i.e. the
  coordinate Frobenii of Definition~\ref{def:coordinate_frobenius}:
  $F_2 \colon A((t^{\mathbb{R}}, u^{\mathbb{R}})) \to A((t^{\mathbb{R}}, u^{\mathbb{R}}))$,
  $t \mapsto t$, $u \mapsto u^\lambda$; and
  $F_3 \colon A((t^{\mathbb{R}}, u^{\mathbb{R}}, v^{\mathbb{R}})) \to
  A((t^{\mathbb{R}}, u^{\mathbb{R}}, v^{\mathbb{R}}))$, fixing $t, u$ and $v \mapsto v^\lambda$.
\end{definition}

\begin{definition}
  \label{def:face_frobenius_action}
  \lean{Novikov.Descent.FM2, Novikov.Descent.FM3}
  \leanok
  \uses{def:face_frobenius, def:descent_datum}
  For $(M, \varphi) \in \mathsf{NovDD}(A)$, the $F_2$-semilinear bijection
  $F_{M,2} = \varphi \circ (\mathrm{id}_M \otimes F_2) \circ \varphi^{-1}$
  on $\pi_2^\ast M$, and the $F_3$-semilinear bijection
  $F_{M,3} = \pi_{13}^\ast\varphi \circ (\mathrm{id}_M \otimes F_3)
  \circ (\pi_{13}^\ast\varphi)^{-1}$ on $\rho_3^\ast M$.
\end{definition}

\begin{lemma}[Cocycle invariance of the three-variable Frobenius]
  \label{lem:fm3_cocycle}
  \lean{Novikov.Descent.FM3_13_eq_FM3_23}
  \leanok
  \uses{def:face_frobenius_action, def:descent_datum}
  $F_{M,3}$ admits the equivalent description via $\pi_{23}^\ast\varphi$ in place
  of $\pi_{13}^\ast\varphi$.
\end{lemma}

\begin{proof}
  \leanok
  Rewrite $\pi_{13}^\ast\varphi = \pi_{23}^\ast\varphi \circ \pi_{12}^\ast\varphi$
  by the cocycle condition and slide $\mathrm{id}_M \otimes F_3$ across
  $\pi_{12}^\ast\varphi$, using that $F_3$ fixes $\pi_{12}^\ast$.
\end{proof}

\begin{theorem}[The descended Frobenius]
  \label{thm:descent_frobenius}
  \lean{Novikov.Descent.descentFrobenius}
  \leanok
  \uses{lem:pi2_module_equalizer, def:face_frobenius_action, lem:fm3_cocycle}
  The maps $\pi_2^\ast, \pi_{13}^\ast, \pi_{23}^\ast$ intertwine
  $F_{M,2}$ and $F_{M,3}$, giving a commutative ladder between the two equalizer
  rows.  Hence $F_{M,2}$ restricts to an $F$-semilinear bijection
  $F_M \colon M \to M$.
\end{theorem}

\begin{proof}
  \leanok
  For $m \in M$, the element $F_{M,2}(1 \otimes m)$ lies in the equalizer:
  its $\pi_{13}^\ast$- and $\pi_{23}^\ast$-images agree because $F_{M,3}$
  intertwines them (Lemma~\ref{lem:fm3_cocycle}).
  Lemma~\ref{lem:pi2_module_equalizer} produces a unique
  $F_M(m) \in M$ with $F_{M,2}(1 \otimes m) = 1 \otimes F_M(m)$; additivity,
  $F$-semilinearity, and invertibility descend from those of $F_{M,2}$.
\end{proof}

\begin{theorem}[The descent-to-isocrystal functor]
  \label{thm:descent_to_isocrystal}
  \lean{Novikov.Descent.descentToIsocrystal}
  \leanok
  \uses{thm:descent_frobenius, lem:pi2_module_equalizer, def:novikov_isocrystal,
    lem:novikov_isocrystal_category}
  The assignment $(M, \varphi) \mapsto (M, F_M)$ defines a functor
  $\mathsf{NovDD}(A) \to \mathsf{NovIsoc}(A)$.
\end{theorem}

\begin{proof}
  \leanok
  Objects go to $(M, F_M)$ via Theorem~\ref{thm:descent_frobenius}; a morphism
  goes to its underlying $A((t^{\mathbb{R}}))$-linear map, which commutes with
  $F_M$ by uniqueness in Lemma~\ref{lem:pi2_module_equalizer}.  Identities and
  composites are preserved because the underlying maps are.
\end{proof}

\begin{proposition}[Constant descent gives constant isocrystals]
  \label{prop:descent_isoc_constant_compat}
  \lean{Novikov.Descent.descentToIsocrystal_comp_vectToNovikovDescent_iso,
    Novikov.Descent.descentToIsocrystal_vectToNovikovDescent_obj_iso}
  \leanok
  \uses{thm:descent_to_isocrystal, def:constant_descent_functor,
    thm:vect_to_isoc_full_faith}
  For a finite projective $A$-module $P$, the isocrystal associated to the
  constant Novikov descent datum on $P$ (Theorem~\ref{thm:descent_to_isocrystal})
  is naturally isomorphic to the constant Novikov isocrystal on $P$.
\end{proposition}

\begin{proof}
  \leanok
  Objectwise both functors give $A((t^{\mathbb{R}})) \otimes_A P$, on which the
  descended Frobenius $F_M = F \otimes \mathrm{id}$ is the Frobenius of the
  constant isocrystal on $P$; the comparison map is thus an isomorphism.
  (The base-change and global $A$-algebra structures on $A((t^{\mathbb{R}}))$ are
  only propositionally equal, but that suffices to identify the two tensors.)
  Naturality holds on pure tensors.
\end{proof}

\subsection{Full faithfulness of the descent-to-isocrystal functor}

\begin{lemma}[Lemma~4.1]
  \label{lem:recover_dd_from_isoc}
  \lean{Novikov.Descent.recoverDDFromIsoc_equalizerLemma,
    Novikov.Descent.RecoverDDFromIsocData}
  \leanok
  \uses{def:novikov_descent_datum, def:face_frobenius_action, lem:fm3_cocycle}
  For a real Novikov descent datum $(M, \varphi)$, consider the two rows
  \[
    M \xrightarrow{\;\varphi \circ \pi_1^\ast\;} \pi_2^\ast M
      \overset{\mathrm{id}}{\underset{F_{M,2}}{\rightrightarrows}} \pi_2^\ast M,
    \qquad
    \pi_2^\ast M \xrightarrow{\;\pi_{23}^\ast\varphi \circ \pi_{12}^\ast\;}
      \rho_3^\ast M \overset{\mathrm{id}}{\underset{F_{M,3}}{\rightrightarrows}}
      \rho_3^\ast M,
  \]
  joined by the vertical maps $\varphi \circ \pi_1^\ast$ (left column) and
  $\pi_{13}^\ast$ (middle and right columns).  The resulting squares commute
  (left square and both right squares $\pi_{13}^\ast \circ F_{M,2} = F_{M,3}
  \circ \pi_{13}^\ast$), and both rows are equalizers: every $F_{M,2}$-fixed
  element of $\pi_2^\ast M$ is $\varphi(\pi_1^\ast m)$ for a unique $m \in M$,
  and every $F_{M,3}$-fixed element of $\rho_3^\ast M$ is
  $\pi_{23}^\ast\varphi(\pi_{12}^\ast x)$ for a unique $x \in \pi_2^\ast M$.
\end{lemma}

\begin{proof}
  \leanok
  The left square is the cocycle condition; the right squares hold by definition
  of $F_{M,3}$ (Lemma~\ref{lem:fm3_cocycle}).  Since $F_{M,2} = F_M \otimes F_2$,
  the top row being an equalizer reduces along $\pi_1^\ast$ and finite
  projectivity of $M$ to the equalizer of $\mathrm{id}$ against $F_2$ on
  $A((t^{\mathbb{R}}, u^{\mathbb{R}}))$, which is the ring-level fixed-point
  statement.  The bottom row is treated the same way via $\pi_{12}^\ast$ and
  $F_3$.
\end{proof}

\begin{lemma}[Isocrystal morphisms preserve descent data]
  \label{lem:descent_to_isoc_full_aux}
  \lean{Novikov.Descent.descentToIsocrystal_full_aux}
  \leanok
  \uses{lem:recover_dd_from_isoc, thm:descent_frobenius}
  Let $(M, \varphi_M), (N, \varphi_N) \in \mathsf{NovDD}(A)$ and let
  $f \colon M \to N$ be $A((t^{\mathbb{R}}))$-linear with $f \circ F_M = F_N
  \circ f$.  Then $\pi_2^\ast f \circ \varphi_M = \varphi_N \circ \pi_1^\ast f$,
  i.e. $f$ is a descent-datum morphism.
\end{lemma}

\begin{proof}
  \leanok
  Apply Lemma~\ref{lem:recover_dd_from_isoc} to $M$ and $N$.  Because $f$
  commutes with the Frobenii and the right squares involve only $F_M, F_N$, the
  equalizer universal properties produce unique $A((t^{\mathbb{R}}))$-linear
  $g \colon M \to N$ and $h \colon \pi_2^\ast M \to \pi_2^\ast N$ filling the
  recovery cube.  Commutativity of the top face shows the bottom face commutes
  with $\pi_2^\ast g$ in place of $h$, so $h = \pi_2^\ast g$ by uniqueness;
  comparing with the left face gives $h = \pi_2^\ast f$, since
  $\varphi_M \circ \pi_1^\ast$ generates $\pi_2^\ast M$.  As $\pi_2^\ast$ is
  injective and $M, N$ are finite projective, $f = g$, and unraveling the top
  face yields $\pi_2^\ast f \circ \varphi_M = \varphi_N \circ \pi_1^\ast f$.
\end{proof}

\begin{proposition}[Proposition~4.2]
  \label{prop:descent_to_isoc_fully_faithful}
  \lean{Novikov.Descent.descentToIsocrystal_fullyFaithful}
  \leanok
  \uses{thm:descent_to_isocrystal, lem:descent_to_isoc_full_aux}
  For every commutative ring $A$, the functor
  $\mathsf{NovDD}(A) \to \mathsf{NovIsoc}(A)$ of
  Theorem~\ref{thm:descent_to_isocrystal} is fully faithful.
\end{proposition}

\begin{proof}
  \leanok
  Faithfulness is immediate: the functor sends a morphism to its underlying
  $A((t^{\mathbb{R}}))$-linear map.  For fullness, an isocrystal morphism
  $\operatorname{descentToIsoc}(M) \to \operatorname{descentToIsoc}(N)$
  is an $A((t^{\mathbb{R}}))$-linear $f \colon M \to N$ commuting with the
  descended Frobenii; Lemma~\ref{lem:descent_to_isoc_full_aux} upgrades it to a
  descent-datum morphism, which the functor sends back to $f$.
\end{proof}
