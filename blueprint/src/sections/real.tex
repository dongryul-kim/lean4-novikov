\section{Real exponents over a general commutative ring}
\label{sec:real_novikov_descent}

We now remove the reducedness hypothesis.  The main point is to find a
\emph{finitely generated} nilpotent ideal modulo which a given descent datum is
constant.  Finiteness is essential: the nilradical itself need not be
nilpotent.

\begin{lemma}[Descent through a nilpotent thickening]
  \label{lem:novikov_descent_nilpotent}
  \lean{Novikov.Descent.novikovDescent_squareZero_of_surjective,
    Novikov.Descent.novikovDescent_baseChangeCompIso,
    Novikov.Descent.novikovDescent_nilpotent}
  \leanok
  \uses{lem:novikov_descent_square_zero}
  Let $J\subseteq A$ be a nilpotent ideal and let $M$ be a real Novikov
  descent datum over $A$.  If $M_{A/J}$ is constant, then $M$ is constant.
\end{lemma}

\begin{proof}
  \leanok
  If $J^N=0$, apply the square-zero deformation theorem successively along
  \[
    A/J\longleftarrow A/J^2\longleftarrow\cdots
      \longleftarrow A/J^N=A.
  \]
  Indeed, the kernel of $A/J^{r+1}\to A/J^r$ is $J^r/J^{r+1}$, whose square
  is zero.  Tensor-product associativity identifies successive coefficient
  extensions with extension along the composite map, so constancy lifts at
  every stage.
\end{proof}

\begin{lemma}[Killing the nonpositive Frobenius-error coefficients]
  \label{lem:nilpotent_positive_frobenius_errors}
  \lean{Novikov.Descent.exists_nilpotent_ideal_coefficients_positive}
  \leanok
  \uses{def:novikov_series, thm:novikov_ring}
  Let $E$ be a finite set and let
  $c_{\alpha\beta}\in A((t^{\mathbb{R}}))$ for $\alpha,\beta\in E$.  Suppose
  that every $c_{\alpha\beta}$ vanishes after coefficient reduction to
  $A/\sqrt{(0)}$.  Then there is a nilpotent ideal $J\subseteq A$ such that
  every image $\overline c_{\alpha\beta}$ in
  $(A/J)((t^{\mathbb{R}}))$ is supported in strictly positive degrees.
\end{lemma}

\begin{proof}
  \leanok
  Let $S\subseteq A$ consist of the nonzero coefficients in degrees at most
  zero among the finitely many series $c_{\alpha\beta}$.  Novikov finiteness
  makes $S$ finite.  Since each series vanishes over $A/\sqrt{(0)}$, every
  element of $S$ belongs to the nilradical.  Let $J$ be the ideal generated by
  $S$.  It is finitely generated and contained in the nilradical, hence is
  nilpotent.  Modulo $J$, all coefficients of every $c_{\alpha\beta}$ in
  degrees at most zero vanish, which is the required positivity.
\end{proof}

\begin{lemma}[Constructing the positive-error lattice]
  \label{lem:positive_frobenius_error_lattice}
  \lean{Novikov.Descent.exists_lattice_frobenius_error_positive}
  \leanok
  \uses{def:novikov_isocrystal, def:frobenius, thm:novikov_module}
  Let $X$ be a Novikov isocrystal over $A$, generated by a finite family
  $(g_\alpha)_{\alpha\in E}$ satisfying
  \[
    F_X(g_\alpha)-g_\alpha
      =\sum_\beta c_{\alpha\beta}g_\beta.
  \]
  Let $J\subseteq A$ be an ideal such that every image
  $\overline c_{\alpha\beta}$ in $(A/J)((t^{\mathbb{R}}))$ is supported in
  strictly positive degrees.  Put $C=A/J$.  Then $X_C$ has a finite lattice
  $L$ such that
  \[
    F_{X_C}(x)-x\in\mathcal O_C^+L
    \qquad(x\in L),
  \]
  where $\mathcal O_C=F_0C((t^{\mathbb{R}}))$ and
  $\mathcal O_C^+$ is its ideal of strictly positive series.
\end{lemma}

\begin{proof}
  \leanok
  Let $\overline g_\alpha$ be the image of $g_\alpha$ in $X_C$ and set
  \[
    L=\sum_\alpha\mathcal O_C\,\overline g_\alpha.
  \]
  The family is finite and still spans after base change, so $L$ is a finite
  lattice in $X_C$.  The base-changed error relation and positivity of the
  $\overline c_{\alpha\beta}$ put
  $F_{X_C}(\overline g_\alpha)-\overline g_\alpha$ in
  $\mathcal O_C^+L$.

  The assertion extends from the generators to their $\mathcal O_C$-span by
  additivity and the identity
  \[
    F_{X_C}(ax)-ax
      =F(a)\bigl(F_{X_C}(x)-x\bigr)+(F(a)-a)x.
  \]
  Indeed, $F(a)$ remains in $\mathcal O_C$, while
  $F(a)-a\in\mathcal O_C^+$ for every $a\in\mathcal O_C$.
\end{proof}

\begin{lemma}[A positive-error lattice gives a constant isocrystal]
  \label{lem:positive_error_lattice_isocrystal_const}
  \lean{Novikov.Descent.isocrystal_const_of_positive_lattice}
  \leanok
  \uses{prop:lattice_isocrystal}
  Let $X$ be a Novikov isocrystal over $A$, let $J\subseteq A$, and put
  $C=A/J$.  Suppose that $X_C$ has a finite lattice $L$ satisfying
  \[
    F_{X_C}(x)-x\in\mathcal O_C^+L
    \qquad(x\in L).
  \]
  Then $X_C$ is isomorphic to the constant isocrystal of a finite projective
  $C$-module.
\end{lemma}

\begin{proof}
  \leanok
  For each $x\in L$, membership of its Frobenius error in
  $\mathcal O_C^+L$ supplies some $\varepsilon>0$ for which
  \[
    F_{X_C}(x)-x\in t^\varepsilon L.
  \]
  Proposition~\ref{prop:lattice_isocrystal} applies to $L$ and gives the
  claimed constant isocrystal.
\end{proof}

\begin{lemma}[Constancy after a nilpotent quotient]
  \label{lem:novikov_descent_const_mod_nilpotent}
  \lean{Novikov.Descent.exists_nilpotent_ideal_descent_const}
  \leanok
  \uses{thm:novikov_descent_equivalence_reduced,
    lem:descent_isocrystal_base_change,
    prop:descent_isoc_constant_compat,
    prop:descent_to_isoc_fully_faithful,
    lem:nilpotent_positive_frobenius_errors,
    lem:positive_frobenius_error_lattice,
    lem:positive_error_lattice_isocrystal_const}
  Let $M$ be a real Novikov descent datum over a commutative ring $A$.  There
  are a nilpotent ideal $J\subseteq A$ and a finite projective $A/J$-module
  $P$ such that
  \[
    \operatorname{Const}(P)\cong M_{A/J}.
  \]
\end{lemma}

\begin{proof}
  \leanok
  Let $X$ be the isocrystal associated to $M$, using the Frobenius with
  $\lambda=2$.  Put $I=\sqrt{(0)}$ and $B=A/I$.  Since $B$ is reduced, the
  reduced case makes $X_B$ a constant isocrystal.

  Lift finite constant generators of $X_B$ to elements $m_i\in X$.  If they do
  not generate $X$, adjoin corrected generators
  \[
    h_j=u_j-\sum_i a_{ji}m_i
  \]
  whose reductions vanish.  The resulting finite family $(g_\alpha)_\alpha$
  generates $X$, and every $F_X(g_\alpha)-g_\alpha$ vanishes over $B$.
  Projectivity of $X$ gives
  \[
    \ker(X\to X_B)
      =\ker\bigl(A((t^{\mathbb{R}}))\to B((t^{\mathbb{R}}))\bigr)X,
  \]
  so there are coefficients, themselves vanishing over $B$, with
  \[
    F_X(g_\alpha)-g_\alpha
      =\sum_\beta c_{\alpha\beta}g_\beta.
  \]

  Lemma~\ref{lem:nilpotent_positive_frobenius_errors} supplies a nilpotent
  ideal $J$ after whose quotient all $\overline c_{\alpha\beta}$ are strictly
  positive.  Lemma~\ref{lem:positive_frobenius_error_lattice} constructs the
  corresponding positive-error lattice in $X_{A/J}$, and
  Lemma~\ref{lem:positive_error_lattice_isocrystal_const} shows that this
  isocrystal is constant.

  Base-change compatibility identifies $X_{A/J}$ with the isocrystal associated
  to $M_{A/J}$.  Compatibility with constant objects and full faithfulness of
  the descent-to-isocrystal functor lift the isocrystal isomorphism to
  $\operatorname{Const}(P)\cong M_{A/J}$ for a finite projective
  $A/J$-module $P$.
\end{proof}

\begin{theorem}[Proposition~4.8: Novikov descent over arbitrary rings]
  \label{thm:novikov_descent_equivalence_real}
  \lean{Novikov.Descent.vectToNovikovDescent_isEquivalence_real,
    Novikov.Descent.vectToNovikovDescent_equivalence_real}
  \leanok
  \uses{prop:vect_to_novikov_descent_fully_faithful,
    lem:novikov_descent_const_mod_nilpotent,
    lem:novikov_descent_nilpotent}
  For every commutative ring $A$, the constant-descent functor
  \[
    \mathsf{Vect}(A)\longrightarrow\mathsf{NovDD}(A,\mathbb{R})
  \]
  is an equivalence of categories.
\end{theorem}

\begin{proof}
  \leanok
  The preceding lemma makes every real Novikov descent datum constant after
  quotienting by some nilpotent ideal.  Lemma~\ref{lem:novikov_descent_nilpotent}
  lifts this constancy back to $A$, proving essential surjectivity.  Full
  faithfulness is
  Proposition~\ref{prop:vect_to_novikov_descent_fully_faithful}.
\end{proof}
