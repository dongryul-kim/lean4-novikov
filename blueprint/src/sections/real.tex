\section{The case of a general commutative ring}
\label{sec:real_novikov_descent}

We now remove the reducedness hypothesis.  The main point is to find a
\emph{finitely generated} nilpotent ideal modulo which a given descent datum is
constant.  Finiteness is essential: the nilradical itself need not be
nilpotent.

\begin{lemma}[Descent through a nilpotent thickening]
  \label{lem:novikov_descent_nilpotent}
  \lean{Novikov.Descent.novikovDescent_squareZero_of_surjective,
    Novikov.Descent.novikovDescent_baseChangeCompIso,
    Novikov.Descent.novikovDescent_nilpotent}
  \leanok
  \uses{lem:novikov_descent_square_zero}
  Let $J\subseteq A$ be a nilpotent ideal and let $M$ be a real Novikov
  descent datum over $A$.  If $M_{A/J}$ is constant, then $M$ is constant.
\end{lemma}

\begin{proof}
  \leanok
  If $J^N=0$, apply the square-zero deformation theorem successively along
  \[
    A/J\longleftarrow A/J^2\longleftarrow\cdots
      \longleftarrow A/J^N=A.
  \]
  Indeed, the kernel of $A/J^{r+1}\to A/J^r$ is $J^r/J^{r+1}$, whose square
  is zero.  Tensor-product associativity identifies successive coefficient
  extensions with extension along the composite map, so constancy lifts at
  every stage.
\end{proof}

\begin{lemma}[Constancy after a nilpotent quotient]
  \label{lem:novikov_descent_const_mod_nilpotent}
  \lean{Novikov.Descent.exists_nilpotent_ideal_descent_const}
  \leanok
  \uses{thm:novikov_descent_equivalence_reduced,
    lem:descent_isocrystal_base_change,
    prop:descent_isoc_constant_compat,
    prop:descent_to_isoc_fully_faithful,
    prop:lattice_isocrystal}
  Let $M$ be a real Novikov descent datum over a commutative ring $A$.  There
  are a nilpotent ideal $J\subseteq A$ and a finite projective $A/J$-module
  $P$ such that
  \[
    \operatorname{Const}(P)\cong M_{A/J}.
  \]
\end{lemma}

\begin{proof}
  \leanok
  Let $X$ be the isocrystal associated to $M$, using the Frobenius with
  $\lambda=2$.  Put $I=\sqrt{(0)}$ and $B=A/I$.  Since $B$ is reduced, the
  reduced case gives a finite projective $B$-module $N_0$ and an isomorphism
  \[
    X_B\cong N_0\otimes_B B((t^{\mathbb R})).
  \]

  Choose generators of $N_0$ and lift the corresponding constant elements to
  $m_1,\ldots,m_n\in X$.  These lifts need not generate $X$, so also choose
  generators $u_1,\ldots,u_k$ of $X$ over $A((t^{\mathbb R}))$.  Write each
  $\overline u_j$ as a linear combination of the $\overline m_i$, lift the
  coefficients, and set
  \[
    h_j=u_j-\sum_i a_{ji}m_i.
  \]
  Then every $h_j$ reduces to zero, while
  \[
    (g_\alpha)_\alpha=(m_1,\ldots,m_n,h_1,\ldots,h_k)
  \]
  still generates $X$.  The reductions of the $m_i$ are Frobenius-fixed, and
  the $h_j$ reduce to zero; hence every
  $F_X(g_\alpha)-g_\alpha$ vanishes over $B$.

  The kernel of module reduction is the product of $X$ with the kernel of
  $A((t^{\mathbb R}))\to B((t^{\mathbb R}))$.  We may therefore write
  \[
    F_X(g_\alpha)-g_\alpha
      =\sum_\beta c_{\alpha\beta}g_\beta,
  \]
  where every $c_{\alpha\beta}$ vanishes over $B$.  Let $J\subseteq A$ be
  generated by the nonzero coefficients of the finitely many
  $c_{\alpha\beta}$ occurring in degrees at most zero.  This is a finite set
  by the Novikov finiteness condition.  Its elements lie in $I$, so $J$ is a
  finitely generated ideal contained in the nilradical and is therefore
  nilpotent.

  Set $C=A/J$.  After reduction to $C$, every
  $c_{\alpha\beta}$ is supported in strictly positive degrees.  Let
  $\mathcal O_C$ be the filtration-zero Novikov power-series ring and set
  \[
    L=\sum_\alpha \mathcal O_C\,\overline g_\alpha\subseteq X_C.
  \]
  This is a finite lattice spanning $X_C$, and its generators satisfy
  \[
    F_{X_C}(\overline g_\alpha)-\overline g_\alpha
      \in \mathcal O_C^+L,
  \]
  where $\mathcal O_C^+$ is the ideal of strictly positive series.  The same
  holds for every $x\in L$, since
  \[
    F_{X_C}(ax)-ax
      =F(a)\bigl(F_{X_C}(x)-x\bigr)+(F(a)-a)x
  \]
  and $F(a)-a\in\mathcal O_C^+$ for $a\in\mathcal O_C$.  Membership in
  $\mathcal O_C^+L$ implies membership in $t^\varepsilon L$ for some
  $\varepsilon>0$.  The lattice criterion now shows that $X_C$ is constant.

  Base-change compatibility identifies $X_C$ with the isocrystal associated
  to $M_C$.  Compatibility with constant objects and full faithfulness of the
  descent-to-isocrystal functor lift this isocrystal isomorphism to
  \(\operatorname{Const}(P)\cong M_C\) for a finite projective $C$-module
  $P$.
\end{proof}

\begin{theorem}[Proposition~4.8: Novikov descent over arbitrary rings]
  \label{thm:novikov_descent_equivalence_real}
  \lean{Novikov.Descent.vectToNovikovDescent_isEquivalence_real,
    Novikov.Descent.vectToNovikovDescent_equivalence_real}
  \leanok
  \uses{prop:vect_to_novikov_descent_fully_faithful,
    lem:novikov_descent_const_mod_nilpotent,
    lem:novikov_descent_nilpotent}
  For every commutative ring $A$, the constant-descent functor
  \[
    \mathsf{Vect}(A)\longrightarrow\mathsf{NovDD}(A,\mathbb R)
  \]
  is an equivalence of categories.
\end{theorem}

\begin{proof}
  \leanok
  The preceding lemma makes every real Novikov descent datum constant after
  quotienting by some nilpotent ideal.  Lemma~\ref{lem:novikov_descent_nilpotent}
  lifts this constancy back to $A$, proving essential surjectivity.  Full
  faithfulness is
  Proposition~\ref{prop:vect_to_novikov_descent_fully_faithful}.
\end{proof}
