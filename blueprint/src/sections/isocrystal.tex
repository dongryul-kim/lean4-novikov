
\section{Novikov isocrystals}
\label{sec:novikov_isocrystals}

In this section we define Novikov isocrystals and prove some basic facts about
them. Throughout the discussion, we fix a real number $\lambda >
1$.

\subsection{Novikov isocrystals}

\begin{definition}[Coordinate Frobenius]
  \label{def:coordinate_frobenius}
  \lean{Novikov.coordinateFrobeniusRingHom}
  \leanok
  \uses{def:novikov_series, thm:novikov_ring}
  Let $R$ be a commutative ring, $\iota$ a finite index set, and
  $j\in\iota$ a coordinate.  The \emph{coordinate Frobenius}
  $F_j \colon R((t_\iota^{\mathbb{R}})) \to R((t_\iota^{\mathbb{R}}))$ is the
  $R$-algebra endomorphism of the multivariable Novikov ring that scales the
  $j$-th exponent by $\lambda$: it sends $t^d \mapsto t^{d'}$ where
  $d'_j = \lambda d_j$ and $d'_i = d_i$ for $i \neq j$.  It is a ring
  homomorphism, and an isomorphism whose inverse is the coordinate Frobenius for
  $1/\lambda$.
\end{definition}

\begin{definition}[One-variable Frobenius]
  \label{def:frobenius}
  \lean{Novikov.frobenius}
  \leanok
  \uses{def:novikov_series, def:coordinate_frobenius}
  For an abelian group $A$, the Frobenius endomorphism $F \colon
  A((t^{\mathbb{R}})) \to A((t^{\mathbb{R}}))$ is the $A$-linear extension of
  $t^d \mapsto t^{\lambda d}$.  It is the one-variable case ($\iota = \ast$)
  of the coordinate Frobenius of Definition~\ref{def:coordinate_frobenius}.
\end{definition}

\begin{theorem}[The Frobenius is a continuous ring homomorphism]
  \label{thm:frobenius_properties}
  \lean{Novikov.frobeniusRingHom, Novikov.frobenius_continuous}
  \leanok
  \uses{def:frobenius, thm:novikov_ring, thm:novikov_topology}
  When $A$ is a ring, the Frobenius is a continuous ring homomorphism.
\end{theorem}

\begin{proof}
  \leanok
  Scaling exponents commutes with their addition, so Frobenius preserves
  convolution.  Moreover, it sends the filtration subgroup $F_D$ into
  $F_{\lambda D}$; this proves continuity with respect to the standard
  neighborhood basis.
\end{proof}

\begin{definition}[Definition~3.1: Novikov isocrystals]
  \label{def:novikov_isocrystal}
  \lean{Novikov.NovikovIsocrystal}
  \leanok
  \uses{def:frobenius, thm:novikov_ring, thm:frobenius_properties}
  Let $A$ be a commutative ring. A \emph{Novikov isocrystal} over $A$ consists of
  \begin{itemize}
    \item a finite projective $A((t^{\mathbb{R}}))$-module $M$, and
    \item a $F$-semilinear isomorphism $F_M \colon M \to M$.
  \end{itemize}
\end{definition}

\begin{lemma}[The category of Novikov isocrystals]
  \label{lem:novikov_isocrystal_category}
  \lean{Novikov.NovikovIsocrystal.Hom,
    Novikov.NovikovIsocrystal.instCategory,
    Novikov.NovikovIsocrystal.instPreadditive,
    Novikov.NovikovIsocrystal.internalHom}
  \leanok
  \uses{def:novikov_isocrystal}
  Novikov isocrystals over $A$ form a preadditive category with internal Hom.
\end{lemma}

\begin{proof}
  \leanok
  Given two isocrystals $M$ and $N$, the module
  $\operatorname{Hom}_{A((t^{\mathbb{R}}))}(M, N)$ is finite projective and
  inherits a Frobenius action from those on $M$ and $N$.
\end{proof}

\begin{theorem}[Proposition~3.3: full faithfulness of constant isocrystals]
  \label{thm:vect_to_isoc_full_faith}
  \lean{Novikov.NovikovIsocrystal.vectToNovIsoc_fullyFaithful}
  \leanok
  \uses{def:novikov_isocrystal, lem:novikov_isocrystal_category}
  The functor $\mathsf{Vect}(A)\to\mathsf{NovIsoc}(A)$ sending a finite
  projective $A$-module $M$ to its constant isocrystal
  $M((t^{\mathbb{R}}))=M\otimes_A A((t^{\mathbb{R}}))$ is fully faithful.
\end{theorem}

\begin{proof}
  \leanok
  A morphism $g$ of constant isocrystals corresponds to an
  $A((t^{\mathbb{R}}))$-linear map
  \[
    M\otimes_A A((t^{\mathbb{R}}))
      \longrightarrow N\otimes_A A((t^{\mathbb{R}}))
  \]
  commuting with $\mathrm{id}\otimes F$.  Via the internal Hom, this yields an
  element of
  $A((t^{\mathbb{R}}))\otimes_A\operatorname{Hom}_A(M,N)$ fixed by $F$.
  Under the base-change identification, write it as an
  $\operatorname{Hom}_A(M,N)$-valued Novikov series $h$.  Frobenius invariance
  gives $h_d=h_{d/\lambda}$ for every $d$.  If a coefficient with $d\ne0$ were
  nonzero, the infinite orbit $d,d/\lambda,d/\lambda^2,\ldots$ would contradict
  Novikov finiteness.  Thus only the degree-zero coefficient remains, and the
  fixed elements are exactly $1\otimes f$ for
  $f\in\operatorname{Hom}_A(M,N)$.
\end{proof}

\subsection{Some lemmas}

\begin{lemma}[A fixed point for the contraction operator]
  \label{lem:contraction_fixed_point}
  \lean{Novikov.contractionOperator,
    Novikov.contractionOperator_estimate,
    Novikov.contractionOperator_fixed}
  \leanok
  \uses{def:frobenius, def:novikov_filtration, thm:novikov_topology,
    thm:novikov_field}
  Let $K$ be a field, let $a_0,\ldots,a_{n-1}\in K((t^{\mathbb{R}}))$ lie in
  the zero filtration, and let $c\in K^\times$.  Define
  \[
    g(x)=\sum_{i=0}^{n-1}
      c^{-(n-i)}F^{n-i}(x)F^{n-1-i}(a_i).
  \]
  If $g(1)$ lies in the zero filtration and has constant coefficient $1$, then
  there is a series $b$ such that $g(b)=b$ and $b_0=1$.
\end{lemma}

\begin{proof}
  \leanok
  The key estimate is
  \[
    x-y\in F_D,\quad D\geq0
    \qquad\Longrightarrow\qquad
    g(x)-g(y)\in F_{\lambda D}.
  \]
  Each summand follows from the facts that $F^k(F_D)\subseteq
  F_{\lambda^kD}$, that $n-i\geq1$, and that every $a_i$ and constant scalar
  lies in $F_0$.

  If $g(1)=1$, take $b=1$.  Otherwise $g(1)-1$ has no terms of degree at most
  zero, so it belongs to $F_\varepsilon$ for some $\varepsilon>0$.  Set
  $b_j=g^j(1)$.  Iterating the estimate gives
  \[
    b_{j+1}-b_j\in F_{\lambda^j\varepsilon}.
  \]
  Since $\lambda^j\varepsilon\to+\infty$, the sequence $(b_j)$ is Cauchy and
  converges by completeness of the Novikov ring.  The same estimate shows
  that $g$ is continuous, so the limit $b$ satisfies $g(b)=b$.  Every $b_j$
  has constant coefficient $1$; convergence in the filtration eventually
  identifies the degree-zero coefficients of $b_j$ and $b$, so $b_0=1$.
\end{proof}

\begin{lemma}[A contraction fixed point gives an eigenvector]
  \label{lem:fixed_point_gives_frobenius_eigenvector}
  \lean{Novikov.bSeq,
    Novikov.bSeq_last_eq_contractionOperator,
    Novikov.exists_frobenius_eigenvector_of_fixed_point}
  \leanok
  \uses{def:novikov_isocrystal, lem:contraction_fixed_point}
  Let $M$ be a Novikov isocrystal over an algebraically closed field $K$.
  Suppose that $m\ne0$, that $n\geq1$, and that
  $m,F_Mm,\ldots,F_M^{n-1}m$ are linearly independent with
  \[
    F_M^nm=\sum_{i=0}^{n-1}a_iF_M^im.
  \]
  Let $c\in K^\times$, let $g$ be the contraction operator formed from the
  $a_i$ and $c$, and suppose that a nonzero series $b$ satisfies $g(b)=b$.
  Then $M$ has a nonzero vector $m'$ satisfying $F_M(m')=c\,m'$.
\end{lemma}

\begin{proof}
  \leanok
  Define a sequence of coefficient series by
  \[
    \beta_0=c^{-1}F(b)a_0,
    \qquad
    \beta_{k+1}=c^{-1}\bigl(F(\beta_k)+F(b)a_{k+1}\bigr).
  \]
  Expanding the recurrence shows that its last term is the contraction:
  $\beta_{n-1}=g(b)=b$.  Hence
  \[
    m'=\sum_{i=0}^{n-1}\beta_iF_M^im
  \]
  is nonzero, because the iterates $F_M^im$ are linearly independent and the
  last coefficient is $b\ne0$.  Apply $F_M$, use semilinearity, substitute the
  relation for $F_M^nm$, and use the recurrence for the $\beta_i$; all terms
  telescope to give $F_M(m')=c\,m'$.
\end{proof}

\begin{theorem}[Existence of a Frobenius eigenvector]
  \label{thm:frobenius_eigenvector}
  \lean{Novikov.exists_frobenius_eigenvector}
  \leanok
  \uses{def:novikov_isocrystal, thm:novikov_field,
    lem:contraction_fixed_point,
    lem:fixed_point_gives_frobenius_eigenvector}
  Let $K$ be an algebraically closed field and $M$ a nonzero Novikov
  isocrystal over $K$. Then there exists a nonzero $m \in M$ and
  $c \in K^\times$ such that $F_M(m) = c \cdot m$.
\end{theorem}

\begin{proof}
  \leanok
  Pick $m_0\ne0$ and choose the minimal $n\geq1$ for which the Frobenius
  iterates satisfy a relation
  \[
    F_M^nm_0=\sum_{i=0}^{n-1}a_iF_M^im_0;
  \]
  the preceding iterates are linearly independent.  Scale $m_0$ by a monomial
  $t^r$ so that the resulting coefficients $a'_i$ lie in the zero filtration
  and at least one constant coefficient $\bar a_i$ is nonzero.  Algebraic
  closedness supplies $c\in K^\times$ satisfying
  \[
    c^n=\sum_{i=0}^{n-1}\bar a_i c^i.
  \]

  Form the contraction operator from the $a'_i$ and $c$.  Direct evaluation
  shows that $g(1)$ lies in the zero filtration; the displayed polynomial
  identity says exactly that its constant coefficient is $1$.
  Lemma~\ref{lem:contraction_fixed_point} therefore gives a fixed point $b$
  with $b_0=1$, in particular $b\ne0$.  The scaled Frobenius iterates remain
  linearly independent and satisfy the relation with coefficients $a'_i$.
  Lemma~\ref{lem:fixed_point_gives_frobenius_eigenvector} now produces the
  required eigenvector.
\end{proof}


\begin{lemma}[Lemma~3.6: Frobenius-stable submodules are coefficientwise]
  \label{lem:frobenius_limit}
  \lean{Novikov.frobenius_limit}
  \leanok
  \uses{def:novikov_series, def:frobenius, def:novikov_filtration,
    thm:novikov_topology, thm:novikov_module}
  Let $A$ be a commutative ring, $M$ an $A$-module, and $S \subseteq
  M((t^{\mathbb{R}}))$ a closed $A$-submodule stable under $F$ and under
  multiplication by every monomial $t^d$.  Then $S = S_0((t^{\mathbb{R}}))$
  where $S_0 = \{ m \in M \mid \text{the constant series } m \in S \}$.
\end{lemma}

\begin{proof}
  \leanok
  First isolate the coefficients of an element $s\in S$.  If $d$ is the least
  exponent in the support of $s$, then
  \[
    F^k(t^{-d}s)\longrightarrow s_d
  \]
  in the Novikov filtration topology: the exponent-zero term stays fixed and
  every other exponent is multiplied by $\lambda^k$.  Stability under shifts
  and Frobenius, followed by closedness, therefore puts the constant series
  $s_d$ in $S$.  For a general exponent $d$, there are only finitely many
  supported exponents below $d$.  Repeatedly isolate and subtract their
  monomials; induction on this finite set reduces to the least-exponent case
  and shows that the constant series $s_d$ lies in $S$.  This proves
  $S\subseteq S_0((t^{\mathbb{R}}))$.

  Conversely, every finite monomial truncation of a series in
  $S_0((t^{\mathbb{R}}))$ lies in $S$ by shift-stability.  These truncations
  converge to the original series, so closedness gives the reverse inclusion.
\end{proof}

\begin{theorem}[Proposition~3.7: injective descent of constancy]
  \label{prop:nov_isoc_injective}
  \lean{Novikov.nov_isoc_injective}
  \leanok
  \uses{def:novikov_isocrystal, lem:frobenius_limit, def:novikov_filtration,
    thm:novikov_topology, def:canonical_topology, thm:canonical_topology_properties,
    thm:projective_novikov_series, thm:novikov_base_change,
    def:frobenius, thm:frobenius_properties, thm:novikov_ring, thm:novikov_module}
  Let $f \colon A \to B$ be an injective ring homomorphism and $M$ a Novikov
  isocrystal over $A$.  If the base change
  $M_B=M\otimes_{A((t^{\mathbb{R}}))}B((t^{\mathbb{R}}))$ is isomorphic to a
  constant isocrystal over $B$, then $M$ is itself isomorphic to a constant isocrystal over $A$.
\end{theorem}

\begin{proof}
  \leanok
  Put $R_A=A((t^{\mathbb{R}}))$ and $R_B=B((t^{\mathbb{R}}))$.  The map
  $R_A\to R_B$ is coefficientwise injective.  Since the underlying
  $R_A$-module of $M$ is finite projective, it is a direct summand of a finite
  free module; the one-tensor map
  \[
    M\longrightarrow R_B\otimes_{R_A}M,
    \qquad m\longmapsto1\otimes m,
  \]
  is therefore a direct summand of the coordinatewise injection
  $R_A^n\to R_B^n$ and is injective.  This argument does not require $R_B$ to
  be flat over $R_A$.

  Through the assumed constant trivialization, this embeds $M$ in
  $N_0((t^{\mathbb{R}}))$.  The coefficient map $R_A\to R_B$ is also a closed
  embedding for the filtration topologies.  Finite projectivity promotes the
  one-tensor map to a closed embedding for the canonical module topologies;
  the constant trivialization is a homeomorphism for those topologies.  Thus
  the image of $M$ is a closed $A$-submodule, stable under Frobenius and every
  monomial shift.  Lemma~\ref{lem:frobenius_limit} identifies it with
  $M_0((t^{\mathbb{R}}))$ for an $A$-submodule $M_0\subseteq N_0$.

  The resulting isomorphism
  $M\cong M_0((t^{\mathbb{R}}))$ makes the latter finite projective over
  $R_A$ and identifies its canonical topology with its Novikov filtration
  topology.  Theorem~\ref{thm:projective_novikov_series} now shows that $M_0$
  is finite projective over $A$.  Frobenius acts coefficientwise under the
  isomorphism, so it identifies $M$ with the constant isocrystal on $M_0$.
\end{proof}

\begin{theorem}[Proposition~3.8: a contracting lattice makes an isocrystal constant]
  \label{prop:lattice_isocrystal}
  \lean{Novikov.lattice_isocrystal}
  \leanok
  \uses{def:novikov_isocrystal, lem:frobenius_limit, def:novikov_filtration,
    thm:novikov_topology, def:canonical_topology,
    thm:canonical_topology_properties, thm:projective_novikov_series,
    lem:split_base_change_descent, thm:novikov_base_change,
    def:frobenius, thm:frobenius_properties, thm:novikov_ring,
    thm:novikov_module}
  Let $P \subseteq A((t^{\mathbb{R}}))$ be the filtration-zero subring and let
  $M$ be a Novikov isocrystal over a commutative ring $A$. Suppose that
  $L \subseteq M$ is a finitely generated $P$-submodule whose
  $A((t^{\mathbb{R}}))$-span is $M$. If for every $m \in L$ there is some
  $\varepsilon>0$ such that
  \[
    F_M(m)-m \in t^\varepsilon L,
  \]
  then $M$ is isomorphic to a constant Novikov isocrystal.
\end{theorem}

\begin{proof}
  \leanok
  A split finite-free presentation of $M$ shows that the shifts
  $t^dL$, $d\in\mathbb{R}$, form a neighborhood basis of zero for the
  canonical topology. Frobenius sends differences in $t^dL$ into
  $t^{\lambda d}L$. Consequently, the Frobenius orbit of every lattice
  element converges to a fixed element congruent to it modulo the ideal of
  positive-degree power series.  This ideal lies in the Jacobson radical of
  $P$, so Nakayama's lemma corrects a finite generating family of $L$ to a
  finite Frobenius-fixed generating family.

  The corrected generators define a Frobenius-compatible surjection
  $(A^n)((t^{\mathbb{R}}))\twoheadrightarrow M$. Its kernel is closed and
  stable under Frobenius and every monomial shift, so
  Lemma~\ref{lem:frobenius_limit} identifies it with
  $K_0((t^{\mathbb{R}}))$ for an $A$-submodule $K_0\subseteq A^n$.
  The splitting of the surjection and the induced topology show, via
  Theorem~\ref{thm:projective_novikov_series}, that $K_0$ is finite
  projective.

  Set $N_0=A^n/K_0$. Exactness of the Novikov-series functor identifies
  $N_0((t^{\mathbb{R}}))$ with $M$, compatibly with Frobenius. The module
  $N_0$ is finitely presented, and split base-change descent along the
  constant-term retraction
  $A((t^{\mathbb{R}}))\to A$ makes it finite projective. Hence the preceding
  identification is an isomorphism from the constant isocrystal attached to
  $N_0$ to $M$.
\end{proof}
