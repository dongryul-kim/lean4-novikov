
\section{Novikov series}

The goal of this section is to define the multivariant Novikov series and prove
basic facts about them.

\subsection{Multi-variable Novikov series}

\begin{definition}[Definition~2.4]
  \label{def:novikov_series}
  \lean{Novikov.NovikovSeries}
  \leanok
  Let $A$ be an abelian group, and let $\Gamma \subseteq \mathbb{R}$ be an
  additive submonoid. We define the abelian group $A((t_1^\Gamma, \dotsc,
  t_n^\Gamma))$ to be the set of power series
  \[ \sum_{i \in I} a_i t_1^{d_{i,1}} \dotsm t_n^{d_{i,n}} \]
  satisfying the condition that $a_i \in A$, $d_i \in \Gamma$, and for all $s_1,
  \dotsc, s_n \in \mathbb{R}_{>0}$ the set $\{ i \in I : \sum_{s_j} d_{i,j} < C
  \}$ is finite.
\end{definition}

\begin{lemma}[Lemma~2.8]
  \label{lem:novikov_exact}
  \lean{Novikov.map_exact}
  \leanok
  \uses{def:novikov_series}
  The functor $A \mapsto A((t_1^\Gamma, \dotsc, t_n^\Gamma))$ is exact. That is,
  if $0 \to A \xrightarrow{f} B \xrightarrow{g} C \to 0$ is an exact sequence of
  abelian groups, then the induced sequence
  \[ 0 \to A((\underline{t}^\Gamma)) \xrightarrow{f} B((\underline{t}^\Gamma))
  \xrightarrow{g} C((\underline{t}^\Gamma)) \to 0 \]
  is also exact.
\end{lemma}

\begin{proof}
  \leanok
  The exactness of the sequence of Novikov series is verified coefficient-wise.
  Since the addition and mapping of series are defined by applying the map to each coefficient
  independently, the injectivity and surjectivity of the induced maps follow from the
  corresponding properties of $f$ and $g$ on the underlying groups.
  Specifically, for surjectivity, any series in $C((\underline{t}^\Gamma))$ can be lifted to
  $B((\underline{t}^\Gamma))$ by lifting each coefficient using the surjectivity of $g$.
  The kernel condition $\text{im}(f) = \text{ker}(g)$ also holds because it holds for each
  coefficient independently.
\end{proof}

\begin{definition}
  \label{def:novikov_multiplication}
  \lean{Novikov.novikovSeriesMul}
  \leanok
  \uses{def:novikov_series}
  Let $A$ and $B$ be abelian groups, and let $\alpha \colon A \times B \to C$ be
  a bilinear map. For $f \in A((t_1^\Gamma, \dotsc, t_n^\Gamma))$ and $g \in
  B((t_1^\Gamma, \dotsc, t_n^\Gamma))$, we define product $f \ast_\alpha g$ by
  bilinearly extending the product
  \[ (a t_1^{d_1} \dotsm t_n^{d_n}) \ast_\alpha (b t_1^{e_1} \dotsm t_n^{e_n})
  = \alpha(a, b) t_1^{d_1+e_1} \dotsm t_n^{d_n+e_n}. \]
\end{definition}

\begin{theorem}
  \label{thm:novikov_ring}
  \lean{Novikov.NovikovSeriesRing}
  \leanok
  \uses{def:novikov_series, def:novikov_multiplication}
  If $A$ is a commutative ring, then the set of Novikov series $A((t_1^\Gamma,
  \dotsc, t_n^\Gamma))$ forms a commutative ring under addition and the
  natural product.
\end{theorem}

\begin{proof}
  \leanok
  The addition is defined coefficient-wise, and the zero element is the series with all zero coefficients.
  The multiplication is defined by convolution: $(f \ast g)_d = \sum_{d_1 + d_2 = d} f_{d_1} g_{d_2}$.
  The finiteness condition on the support of the product follows from the fact that the sum of two
  well-ordered sets (or sets satisfying the Novikov finiteness condition) also satisfies the condition.
  The ring axioms (associativity, distributivity, and existence of unity $1 \cdot t^0$) follow from
  the corresponding properties of the underlying ring $A$ and the properties of formal power series
  summation.
\end{proof}

\begin{theorem}
  \label{thm:novikov_module}
  \lean{Novikov.moduleNovikovSeries}
  \leanok
  \uses{thm:novikov_ring, def:novikov_multiplication}
  If $A$ is a commutative ring and $M$ is an $A$-module, then $M((t_1^\Gamma,
  \dotsc, t_n^\Gamma))$ is naturally a module over the Novikov ring
  $A((t_1^\Gamma, \dotsc, t_n^\Gamma))$.
\end{theorem}

\begin{proof}
  \leanok
  The $A((\underline{t}^\Gamma))$-module structure on $M((\underline{t}^\Gamma))$ is defined via the
  convolution product using the scalar multiplication $\alpha \colon A \times M \to M$.
  The module axioms, such as $(f \ast g) \cdot m = f \cdot (g \cdot m)$ and distributivity,
  are verified by comparing coefficients, which involves reordering sums based on the associativity
  and distributivity of the underlying module $M$ over $A$.
\end{proof}

\begin{theorem}
  \label{thm:novikov_base_change}
  \lean{Novikov.novikovModule_base_change_equiv}
  \leanok
  \uses{thm:novikov_module, lem:novikov_exact}
  Let $M$ be a finitely presented $A$-module. The natural map
  \[ M \otimes_A A((t_1^\Gamma, \dotsc, t_n^\Gamma)) \to M((t_1^\Gamma, \dotsc, t_n^\Gamma)) \]
  is an isomorphism of $A((t_1^\Gamma, \dotsc, t_n^\Gamma))$-modules.
\end{theorem}

\begin{proof}
  \leanok
  Since $M$ is finitely presented, it has a presentation $F \to G \to M \to 0$ with $F, G$ finite free.
  Applying the functor $M \mapsto M((\underline{t}^\Gamma))$ gives a right exact sequence
  $F((\underline{t}^\Gamma)) \to G((\underline{t}^\Gamma)) \to M((\underline{t}^\Gamma)) \to 0$ by Lemma~\ref{lem:novikov_exact}.
  Tensor product also preserves right exactness, giving $F \otimes A((\underline{t}^\Gamma)) \to G \otimes A((\underline{t}^\Gamma)) \to M \otimes A((\underline{t}^\Gamma)) \to 0$.
  The natural map gives a morphism between these two sequences. For finite free modules, the map is an isomorphism.
  By the five lemma (specifically the four lemma variant for right exactness), the map for $M$ is also an isomorphism.
\end{proof}

\begin{definition}
  \label{def:novikov_substitute}
  \lean{Novikov.substitute}
  \leanok
  \uses{def:novikov_series}
  Given a map $f \colon [n] \to [m]$ between finite sets of variables, we
  define the substitution map
  \[ f_\ast \colon A((t_1^\Gamma, \dotsc, t_m^\Gamma)) \to
  A((s_1^\Gamma, \dotsc, s_m^\Gamma)) \]
  given by substituting the variables. When it is clear from the context, we
  shall denote this induced map by $f$ instead of $f_\ast$.
\end{definition}

\begin{theorem}
  \label{thm:novikov_substitute_functorial}
  \lean{Novikov.substitute_id, Novikov.substitute_comp, Novikov.substituteRingHom}
  \leanok
  \uses{def:novikov_substitute, thm:novikov_ring}
  The substitution map is functorial in $\iota$. That is, $\text{id}_* =
  \text{id}$ and $(g \circ f)_* = g_* \circ f_*$. Furthermore, if $A$ is a
  commutative ring, then $f_*$ is a ring homomorphism.
\end{theorem}

\begin{proof}
  \leanok
  Functoriality and the additive property are checked by direct calculation.
  The preservation of multiplication follows from the properties of convolution.
\end{proof}

\subsection{Single-variable Novikov series}

When there is a single variable $t$, we give $A((t^\Gamma))$ a topology.

\begin{definition}
  \label{def:novikov_filtration}
  \lean{Novikov.filtration}
  \leanok
  \uses{def:novikov_series}
  For $D \in \mathbb{R}$, we define the filtration subgroup $F_D A((t^\Gamma))$
  as the set of series $f = \sum_{d \in \Gamma} a_d t^d$ such that $a_d = 0$ for
  all $d < D$.
\end{definition}

\begin{theorem}[Definition~2.15]
  \label{thm:novikov_topology}
  \lean{Novikov.is_topological_add_group, Novikov.is_topological_ring}
  \leanok
  \uses{def:novikov_filtration, thm:novikov_ring}
  The collection of filtration subgroups $\{F_D A((t^\Gamma))\}_{D \in \mathbb{R}}$
  forms a filter basis for a topology on $A((t^\Gamma))$. Under this topology,
  $A((t^\Gamma))$ is a complete topological abelian group. Furthermore, if $A$
  is a commutative ring, then $A((t^\Gamma))$ is a topological ring.
\end{theorem}

\begin{proof}
  \leanok
  The filtration subgroups satisfy $F_{D_2} \subseteq F_{D_1}$ for $D_1 \le D_2$,
  which ensures that the intersection of two filtration subgroups contains
  another. Each $F_D$ is an additive subgroup, so they form a filter basis for a
  topological additive group. For the ring structure, continuity of
  multiplication follows from the inclusion $F_{D_1} \cdot F_{D_2} \subseteq
  F_{D_1 + D_2}$.
\end{proof}

\begin{theorem}
  \label{thm:novikov_field}
  \lean{Novikov.novikovField}
  \leanok
  \uses{thm:novikov_topology}
  If $A$ is a field, then the ring of single-variable Novikov series $A((t^\Gamma))$
  is a field.
\end{theorem}

\begin{proof}
  \leanok
  Every nonzero series $f$ has a well-defined minimum degree $d = \min\{d' \mid f(d') \neq 0\}$.
  We define the inverse $f^{-1}$ by constructing the geometric series for $1 - f \cdot a^{-1} t^{-d}$,
  where $a = f(d)$ is the leading coefficient. Since $f \cdot a^{-1} t^{-d}$ has minimum degree $0$ and
  leading coefficient $1$, its complement is positive and topologically nilpotent, ensuring
  summability of the geometric series.
\end{proof}

\subsection{Canonical module topology}

\begin{definition}
  \label{def:canonical_topology}
  \lean{Novikov.Miscellany.canonicalTopology}
  \leanok
  Let $A$ be a topological commutative ring, and let $M$ be an $A$-module.
  The \emph{canonical topology} on $M$ is the initial topology with respect to
  all $A$-linear maps $M \to A$. That is, it is the coarsest topology on $M$
  for which every $A$-linear map $f \colon M \to A$ is continuous.
\end{definition}

\begin{theorem}
  \label{thm:canonical_topology_properties}
  \lean{Novikov.Miscellany.canonicalTopology.continuousAdd,
    Novikov.Miscellany.canonicalTopology.continuousSMul,
    Novikov.Miscellany.canonicalTopology_self_eq,
    Novikov.Miscellany.canonicalTopology.continuous_linearMap,
    Novikov.Miscellany.canonicalTopology_prod_eq}
  \leanok
  \uses{def:canonical_topology}
  Let $A$ be a topological commutative ring, and let $M$, $N$ be $A$-modules.
  \begin{enumerate}
    \item The canonical topology makes $M$ into a topological $A$-module.
    \item On $A$ itself, the canonical topology coincides with the original
      topology.
    \item Every $A$-linear map $M \to N$ is continuous with respect to the
      canonical topologies.
    \item The canonical topology on $M \times N$ coincides with the product
      topology of the canonical topologies on $M$ and $N$.
  \end{enumerate}
\end{theorem}

\begin{proof}
  \leanok
  (1) For each $f \colon M \to A$ we need to show that $A \times M \to M
  \xrightarrow{f} A$ is continuous. This follows from that this is a composition
  $A \times M \xrightarrow{\mathrm{id} \times f} A \times A \to A$, which is a
  composition of two continuous functions.

  (2) The identity map $\mathrm{id}_A \colon A \to A$ is $A$-linear, so the
  canonical topology on $A$ is finer than the original topology. Conversely,
  every $A$-linear map $f \colon A \to A$ is given by $f(x) = f(1)x$, which is
  continuous in the original topology because $A$ is a topological ring. Hence
  the original topology is finer than each induced topology, so the two
  topologies coincide.

  (3) For any $A$-linear $f \colon N \to A$, the composition $f \circ \varphi
  \colon M \to A$ is $A$-linear and therefore continuous in the canonical
  topology on $M$. Since the canonical topology on $N$ is initial for all such
  $f$, $\varphi$ is continuous.

  (4) Any $A$-linear map $f \colon M \times N \to A$ decomposes as $f(x,y) =
  f_1(x) + f_2(y)$ where $f_1 = f \circ \iota_M$ and $f_2 = f \circ \iota_N$
  are $A$-linear. Both $f_1$ and $f_2$ are continuous in the product of
  canonical topologies, so the product topology is finer than the canonical
  topology. Conversely, the projections $M \times N \to M$ and $M \times N \to
  N$ are $A$-linear and thus continuous in the canonical topology, so the
  canonical topology is finer than the product topology.
\end{proof}

\begin{lemma}
  \label{lem:split_base_change_descent}
  \lean{Novikov.Miscellany.finite_projective_of_split_baseChange}
  \leanok
  Let $B$ be an $A$-algebra with an $A$-linear retraction $B \to A$.
  If $B \otimes_A M$ is a finite projective $B$-module, then $M$ is a
  finite projective $A$-module.
\end{lemma}

\begin{proof}
  \leanok
  Write $\sigma \colon B \to A$ for the $A$-linear retraction. For finiteness,
  we choose a finite set of $B$-module generators of $B \otimes_A M$, and their
  image under $\sigma \otimes \mathrm{id}_M$ will generate $M$ as a $A$-module.
  For projectivity, given a $A$-linear surjection $\varepsilon \colon P
  \twoheadrightarrow Q$ and $M \to Q$, we wish to lift it to $M \to P$. We lift
  its $B$-base change along $\varepsilon_B$ using $B$-projectivity and push it
  down using $\sigma$. The result gives a lift witnessing the $A$-projectivity
  of $M$.
\end{proof}

\begin{theorem}[Lemma~2.16]
  \label{thm:projective_novikov_series}
  \lean{Novikov.projective_of_realNovikovSeries}
  \leanok
  \uses{lem:split_base_change_descent, thm:novikov_base_change, thm:canonical_topology_properties}
  Let $M$ be an $A$-module. If $M ((t^{\mathbb{R}}))$ is a finite projective
  $A ((t^{\mathbb{R}}))$-module and its canonical topology coincides with the
  natural filtration topology, then $M$ is a finite projective $A$-module.
\end{theorem}

\begin{proof}
  \leanok
  First, $M ((t^{\mathbb{R}}))$ is finite projective so the natural surjection
  $R^n \twoheadrightarrow M ((t^{\mathbb{R}}))$ splits. The continuity
  hypothesis implies each coordinate of the splitting lands in the zero
  filtration under a common shift. Using the monomial at height $N$, one
  extracts a finite $A$-spanning set for $M$, proving finite generation.

  For finite presentation, apply the same argument to the kernel of the
  surjection $A^m \twoheadrightarrow M$, noting that the kernel's Novikov
  series identifies with a direct summand of $R^m$ and thus inherits finite
  projectivity and the canonical topology condition.

  Finally, the base change isomorphism
  $M ((t^{\mathbb{R}})) \cong A ((t^{\mathbb{R}})) \otimes_A M$ together with
  the $A$-linear splitting $A ((t^{\mathbb{R}})) \to A$ given by evaluation
  at $0$ reduces to Lemma~\ref{lem:split_base_change_descent}.
\end{proof}
